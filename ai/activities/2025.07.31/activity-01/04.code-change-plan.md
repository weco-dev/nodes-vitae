# Code Change Plan: ESG Assessment Demo Implementation

## 1. Overview of Changes

This plan implements a standalone demo version of the ESG Assessment system that allows unauthenticated users to experience the assessment workflow with 25 questions across all 7 ESG sections. The implementation follows the standalone approach (Option A) with localStorage persistence, maintaining full UI fidelity while avoiding database operations.

### Key Features:
- New `_demo+` route structure with layout, take, and complete screens
- localStorage-based persistence for session data
- 25 demo questions with umbrella question support
- Excel import system for demo questions
- Landing page integration with "Try Demo" CTA
- Analytics tracking for demo usage
- Mock results display upon completion

### Architecture Strategy:
- 90%+ code reuse through component adaptation
- Clean separation between demo and production logic
- No authentication requirements for demo routes
- Full mobile-responsive experience maintained

```typescript
const DEMO_STORAGE_KEY = 'vitae-demo-assessment'
const DEMO_ANALYTICS_KEY = 'vitae-demo-analytics'

interface DemoData {
  currentPageIndex: number
  surveyData: Record<string, any>
  startTime: number
  lastUpdated: number
  isCompleted: boolean
}

interface DemoAnalyticsEvent {
  event: string
  timestamp: number
  data: Record<string, any>
}

export function getDemoDataFromLocalStorage(): DemoData {
  try {
    const stored = localStorage.getItem(DEMO_STORAGE_KEY)
    if (!stored) return getDefaultDemoData()
    return JSON.parse(stored)
  } catch {
    return getDefaultDemoData()
  }
}

export function saveDemoProgressToLocalStorage(surveyData: Record<string, any>, pageIndex: number): void {
  try {
    const current = getDemoDataFromLocalStorage()
    const updated: DemoData = {
      ...current,
      surveyData,
      currentPageIndex: pageIndex,
      lastUpdated: Date.now()
    }
    localStorage.setItem(DEMO_STORAGE_KEY, JSON.stringify(updated))
  } catch (error) {
    console.warn('Failed to save demo progress:', error)
  }
}

export function saveDemoAnswerToLocalStorage(questionName: string, answer: any): void {
  try {
    const current = getDemoDataFromLocalStorage()
    const updated: DemoData = {
      ...current,
      surveyData: {
        ...current.surveyData,
        [questionName]: answer
      },
      lastUpdated: Date.now()
    }
    localStorage.setItem(DEMO_STORAGE_KEY, JSON.stringify(updated))
  } catch (error) {
    console.warn('Failed to save demo answer:', error)
  }
}

export function completeDemoInLocalStorage(): void {
  try {
    const current = getDemoDataFromLocalStorage()
    const updated: DemoData = {
      ...current,
      isCompleted: true,
      lastUpdated: Date.now()
    }
    localStorage.setItem(DEMO_STORAGE_KEY, JSON.stringify(updated))
    trackDemoAnalytics('demo_completed', { totalQuestions: Object.keys(current.surveyData).length })
  } catch (error) {
    console.warn('Failed to complete demo:', error)
  }
}

export function clearDemoData(): void {
  try {
    localStorage.removeItem(DEMO_STORAGE_KEY)
    localStorage.removeItem(DEMO_ANALYTICS_KEY)
  } catch (error) {
    console.warn('Failed to clear demo data:', error)
  }
}

export function trackDemoAnalytics(event: string, data: Record<string, any> = {}): void {
  try {
    const events = getDemoAnalyticsEvents()
    events.push({
      event,
      timestamp: Date.now(),
      data
    })
    localStorage.setItem(DEMO_ANALYTICS_KEY, JSON.stringify(events))
    
    // Also send to monitoring if available
    if (typeof window !== 'undefined' && window.gtag) {
      window.gtag('event', event, { ...data, demo: true })
    }
  } catch (error) {
    console.warn('Failed to track demo analytics:', error)
  }
}

function getDefaultDemoData(): DemoData {
  return {
    currentPageIndex: 0,
    surveyData: {},
    startTime: Date.now(),
    lastUpdated: Date.now(),
    isCompleted: false
  }
}

function getDemoAnalyticsEvents(): DemoAnalyticsEvent[] {
  try {
    const stored = localStorage.getItem(DEMO_ANALYTICS_KEY)
    return stored ? JSON.parse(stored) : []
  } catch {
    return []
  }
}
```

```typescript
export function initializeDemoAnalytics(): void {
  trackDemoAnalytics('demo_started', {
    userAgent: navigator.userAgent,
    timestamp: Date.now(),
    referrer: document.referrer
  })
}

export function trackDemoQuestionAnswer(questionId: string, questionName: string, answer: any): void {
  trackDemoAnalytics('demo_question_answered', {
    questionId,
    questionName,
    answerType: typeof answer,
    hasValue: answer !== null && answer !== undefined && answer !== ''
  })
}

export function trackDemoNavigation(from: number, to: number): void {
  trackDemoAnalytics('demo_navigation', {
    fromPage: from,
    toPage: to,
    direction: to > from ? 'forward' : 'backward'
  })
}

export function trackDemoAbandonment(): void {
  trackDemoAnalytics('demo_abandoned', {
    timestamp: Date.now(),
    sessionDuration: Date.now() - (getDemoDataFromLocalStorage()?.startTime || Date.now())
  })
}

export function trackDemoCompletion(): void {
  const demoData = getDemoDataFromLocalStorage()
  trackDemoAnalytics('demo_completed', {
    sessionDuration: Date.now() - (demoData?.startTime || Date.now()),
    questionsAnswered: Object.keys(demoData?.surveyData || {}).length,
    completionRate: 1.0
  })
}
```

```tsx
export async function loader({ request }: Route.LoaderArgs) {
  // No authentication required for demo

  return {

export async function action({ request }: Route.ActionArgs) {
  const formData = await request.formData()
  const intent = formData.get('intent')

  switch (intent) {
    case 'save-progress': {
      const surveyDataStr = formData.get('surveyData')
      if (typeof surveyDataStr !== 'string') {
        throw new Error('Invalid survey data')
      }
      const surveyData = JSON.parse(surveyDataStr) as Record<string, any>
      const currentPageIndex = Number(formData.get('currentPageIndex'))

      saveDemoProgressToLocalStorage(surveyData, currentPageIndex)
      return { success: true }
    }

    case 'save-answer': {
      const questionName = formData.get('questionName') as string
      const answerStr = formData.get('answer')
      if (typeof answerStr !== 'string') {
        throw new Error('Invalid answer data')
      }
      const answer = JSON.parse(answerStr)

      saveDemoAnswerToLocalStorage(questionName, answer)
      
      // Track analytics for demo
      const questionMeta = JSON.parse(formData.get('questionMeta') as string)
      trackDemoQuestionAnswer(questionMeta.questionId, questionName, answer)
      
      return { success: true }
    }

    case 'complete': {
      completeDemoInLocalStorage()
      return redirect('/_demo/complete')
    }

    default:
      throw new Error(`Unknown intent: ${intent}`)
  }
}

export default function DemoTake() {
  const { assessment, surveyJson, questions } = useLoaderData<typeof loader>()
  const fetcher = useFetcher()

  // Initialize demo analytics on mount
  useEffect(() => {
    initializeDemoAnalytics()
  }, [])

  // Demo-specific navigation system (adapted from assessment route)
  const assessmentNavigation = useAssessmentNavigation(
    assessment.currentPageIndex,
    questions.length,
    async (pageIndex: number, signal?: AbortSignal) => {
      if (signal?.aborted) {
        throw new Error('Navigation aborted')
      }

      // Update survey model
      const surveyModel = (window as any).surveyModel
      if (surveyModel) {
        isProgrammaticNavigation.current = true
        surveyModel.currentPageNo = pageIndex
        setCurrentSection(questions[pageIndex]?.section || '')
        
        // Track navigation for demo analytics
        trackDemoNavigation(assessment.currentPageIndex, pageIndex)
        
        setTimeout(() => {
          isProgrammaticNavigation.current = false
        }, 100)
      }

      if (signal?.aborted) {
        throw new Error('Navigation aborted')
      }

      // Save progress to localStorage instead of database
      return new Promise((resolve, reject) => {
        void fetcher.submit(
          {
            intent: 'save-progress',
            currentPageIndex: String(pageIndex),
            surveyData: JSON.stringify(surveyModel?.data || {}),
          },
          { method: 'POST' },
        )

        const timeoutId = setTimeout(() => {
          reject(new Error('Navigation timeout'))
        }, 2000)

        const checkCompletion = () => {
          if (signal?.aborted) {
            clearTimeout(timeoutId)
            reject(new Error('Navigation aborted'))
            return
          }

          if (fetcher.state === 'idle') {
            clearTimeout(timeoutId)
            resolve(void 0)
          } else {
            setTimeout(checkCompletion, 50)
          }
        }

        checkCompletion()
      })
    },
  )

  const [currentSection, setCurrentSection] = useState(
    questions[assessment.currentPageIndex]?.section || questions[0]?.section || '',
  )
  const [answeredQuestions, setAnsweredQuestions] = useState<Set<string>>(new Set())
  const [loadingTimeout, setLoadingTimeout] = useState(false)
  const isProgrammaticNavigation = useRef(false)

  // Handle loading timeout
  useEffect(() => {
    const timer = setTimeout(() => {
      setLoadingTimeout(true)
    }, 10000)
    return () => clearTimeout(timer)
  }, [])

  // Initialize answered questions from demo data
  useEffect(() => {
    const answered = new Set<string>()
    if (assessment.surveyData && typeof assessment.surveyData === 'object') {
      Object.keys(assessment.surveyData).forEach((fieldName) => {
        const value = assessment.surveyData![fieldName]
        const hasValue = value !== undefined && value !== null && value !== '' && 
                         !((value as any)?.length === 0)

        if (hasValue) {
          const question = questions.find((q) => {
            const fullQuestion = demoAssessmentQuestions.find((aq) => aq.questionId === q.questionId)
            return fullQuestion?.name === fieldName
          })
          if (question) {
            answered.add(question.questionId)
          }
        }
      })
    }
    setAnsweredQuestions(answered)
  }, [questions, assessment.surveyData])

  const handleValueChanged = useCallback(
    (name: string, value: any, questionMeta: any) => {
      // Update answered questions for UI feedback
      const question = questions.find((q) => {
        const fullQuestion = demoAssessmentQuestions.find((aq) => aq.questionId === q.questionId)
        return fullQuestion?.name === name
      })

      // Only process answers for non-umbrella questions
      const fullQuestion = demoAssessmentQuestions.find((aq) => aq.questionId === question?.questionId)
      if (fullQuestion?.type === 'group') {
```
