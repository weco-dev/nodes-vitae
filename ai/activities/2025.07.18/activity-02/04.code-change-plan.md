# Code Change Plan: ESG Assessment UI Improvements

## 1. Overview of Changes

This plan implements all UI improvements for the ESG Assessment platform,
specifically addressing desktop progress bar layout issues, adding section
navigation, implementing auto-scroll behavior, fixing initial page navigation,
and resolving description display problems. The changes affect 3 core components
and 1 route file, transforming the assessment experience from a fragmented
interface to a streamlined, user-friendly navigation system.

## 2. Setup and Configuration

No new npm packages are required. All necessary dependencies (shadcn/ui
components, SurveyJS, React hooks) are already available in the project.

### 3.2. Auto-scroll Behavior Implementation

#### 3.2.1. Add Scroll Management to Progress Bar

- **Files to modify:** app/components/assessment/responsive-progress-bar.tsx
- **Changes to make:**
  - Add useRef for scroll container
  - Implement scrollToCurrentDot function
  - Add useEffect to trigger scroll on currentIndex change
  - Update navigation handlers to trigger scroll
- **Code changes:**

  ```tsx
  // Add imports at the top
  import { useRef, useEffect } from 'react'

  // Add refs inside component
  const mobileScrollRef = useRef<HTMLDivElement>(null)
  const desktopScrollRef = useRef<HTMLDivElement>(null)

  // Add scroll function
  const scrollToCurrentDot = useCallback(() => {
    const scrollContainer = window.innerWidth >= 1024 ? desktopScrollRef.current : mobileScrollRef.current
    if (scrollContainer) {
      const dots = scrollContainer.querySelectorAll('button[aria-label*="Question"]')
      const currentDot = dots[currentIndex] as HTMLElement
      if (currentDot) {
        currentDot.scrollIntoView({
          behavior: 'smooth',
          inline: 'center',
          block: 'nearest'
        })
      }
    }
  }, [currentIndex])

  // Add useEffect for auto-scroll
  useEffect(() => {
    const timer = setTimeout(() => {
      scrollToCurrentDot()
    }, 100) // Small delay to ensure DOM is updated

    return () => clearTimeout(timer)
  }, [currentIndex, scrollToCurrentDot])

  // Update navigation handlers
  const handlePrevious = () => {
    if (currentIndex > 0) {
      onPageChange(currentIndex - 1)
    }
  }

  const handleNext = () => {
    if (currentIndex < questions.length - 1) {
      onPageChange(currentIndex + 1)
    }
  }

  // Add ref to mobile scroll container
  <div ref={mobileScrollRef} className="flex-1 overflow-x-auto">

  // Add ref to desktop scroll container
  <div ref={desktopScrollRef} className="flex-1 overflow-x-auto">
  ```

- **Dependencies:** React useRef, useEffect, useCallback
- **Considerations:** Small delay prevents race conditions with DOM updates
- **Considerations:** Add debouncing logic in take.tsx (to avoid moving wrongly
  between questions when I click the arrow many times fast)
- **Considerations:** hide the scroll bar when changing the question with the
  progress bar

### 3.3. Section Navigation Dropdown Implementation

#### 3.3.1. Create Section Navigation Component

- **Files to modify:** app/components/assessment/section-display.tsx
- **Changes to make:**
  - Replace entire component with Select dropdown
  - See the whole string in the section property
  - Add section navigation logic
  - Extract unique sections from questions
  - Implement section-to-question mapping
- **Code changes:**

  ```tsx
  // Replace entire file content:
  import {
  	Select,
  	SelectContent,
  	SelectItem,
  	SelectTrigger,
  	SelectValue,
  } from '#app/components/ui/select'

  interface SectionDisplayProps {
  	section: string
  	currentQuestion: number
  	totalQuestions: number
  	questions: Array<{
  		questionId: string
  		section: string
  		title: string
  		isRequired: boolean
  	}>
  	onSectionChange: (sectionIndex: number) => void
  }

  export function SectionDisplay({
  	section,
  	currentQuestion,
  	totalQuestions,
  	questions,
  	onSectionChange,
  }: SectionDisplayProps) {
  	// Extract unique sections and their first question indices
  	const sections = questions.reduce(
  		(acc, question, index) => {
  			if (!acc.find((s) => s.section === question.section)) {
  				acc.push({
  					section: question.section,
  					firstQuestionIndex: index,
  					label: question.section.split('|')[0].trim(), // Clean section name
  				})
  			}
  			return acc
  		},
  		[] as Array<{
  			section: string
  			firstQuestionIndex: number
  			label: string
  		}>,
  	)

  	const handleSectionChange = (sectionName: string) => {
  		const selectedSection = sections.find((s) => s.section === sectionName)
  		if (selectedSection) {
  			onSectionChange(selectedSection.firstQuestionIndex)
  		}
  	}

  	return (
  		<div className="bg-muted mb-6 rounded-lg p-4">
  			<div className="flex items-center justify-between">
  				<div className="mr-4 flex-1">
  					<p className="text-muted-foreground mb-2 text-sm">
  						Navigate to Section
  					</p>
  					<Select value={section} onValueChange={handleSectionChange}>
  						<SelectTrigger className="w-full">
  							<SelectValue placeholder="Select section..." />
  						</SelectTrigger>
  						<SelectContent>
  							{sections.map((sectionItem) => (
  								<SelectItem
  									key={sectionItem.section}
  									value={sectionItem.section}
  								>
  									{sectionItem.label}
  								</SelectItem>
  							))}
  						</SelectContent>
  					</Select>
  				</div>
  				<div className="text-right">
  					<p className="text-muted-foreground text-sm">Question</p>
  					<p className="text-lg font-semibold">
  						{currentQuestion} / {totalQuestions}
  					</p>
  				</div>
  			</div>
  		</div>
  	)
  }
  ```

- **Dependencies:** shadcn/ui Select components
- **Considerations:** Maintains existing visual structure while adding
  navigation
